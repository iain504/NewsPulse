   329→  // ---- SSE analysis ----
   330→  const analyzeSymbol = React.useCallback(
   331→    async (sym: string, source: "manual" | "auto", silent: boolean): Promise<void> => {
   332→      if (!silent) {
   333→        setStatus("fetching");
   334→        setSignal(null);
   335→        setRationale("");
   336→        setStreaming(false);
   337→        setAnalysisError(null);
   338→        setLatency(0);
   339→        setSignalId(null);
   340→        setAnalyzedSymbol("");

   343→      let errorMessage = "";

   345→        await postSSE(
   346→          "/api/signals/analyze",
   347→          { symbol: sym, interval: timeframe, model, riskProfile, source },
   348→          {
   349→            onEvent: (event, data) => {
   350→              if (event === "snapshot") {
   351→                if (!silent) {
   352→                  setSnapshot((data as SnapshotEvent).snapshot);
   353→                  setStatus("thinking");
   354→                }
   355→              } else if (event === "signal") {
   356→                const sig = (data as SignalEvent).signal;
   357→                if (!silent) {
   358→                  setSignal(sig);
   359→                  setLatency((data as SignalEvent).latencyMs);
   360→                  setStatus("ready");
   361→                  setAnalyzedKey(`${sym}|${timeframe}`);
   362→                  setAnalyzedSymbol(sym);
   363→                } else if (sig.direction === "BUY" || sig.direction === "SELL") {
   364→                  const headline = `${sym} → ${sig.direction} @ ${formatPrice(sig.entry)}`;
   365→                  const detail = `confidence ${Math.round(sig.confidence * 100)}% · SL ${formatPrice(sig.stopLoss)} · TP ${formatPrice(sig.takeProfit)}`;
   366→                  if (sig.direction === "BUY") toast.success(headline, { description: detail });
   367→                  else toast.warning(headline, { description: detail });
   368→                }
   369→              } else if (event === "delta") {
   370→                if (!silent) {
   371→                  setStreaming(true);
   372→                  setRationale((r) => r + (data as { t: string }).t);
   373→                }
   374→              } else if (event === "done") {
   375→                const doneData = data as { id?: string };
   376→                if (doneData?.id) setSignalId(doneData.id);
   377→                if (!silent) setStreaming(false);
   378→                void refreshFeed();
   379→              } else if (event === "error") {
   380→                errored = true;
   381→                errorMessage = (data as ErrorEvent).message;
   382→                if (!silent) {
   383→                  setAnalysisError(errorMessage);
   384→                  setStatus("error");
   385→                  setStreaming(false);
   386→                }
   387→              }
   388→            },
   389→            onError: (err) => {
   390→              errored = true;
   391→              errorMessage = err.message;
   392→              if (!silent) {
   393→                setAnalysisError(err.message);
   394→                setStatus("error");
   395→              }
   396→            },
   397→          },
   398→        );
   399→      } catch (err) {
   400→        errored = true;
   401→        errorMessage = err instanceof Error ? err.message : "Request failed";
   402→        if (!silent) {
   403→          setAnalysisError(errorMessage);
   404→          setStatus("error");
   405→        }
   406→      }
   407→      if (errored) throw new Error(errorMessage || `${sym}: analysis failed`);
   408→    },
   409→    [timeframe, model, riskProfile, refreshFeed],
   410→  );

   485→        method: "POST",
   486→        headers: { "Content-Type": "application/json" },
   487→        body: JSON.stringify({ symbol: s }),

   489→      setWatchItems((items) => [...items, res.item]);
   490→      setCustomSymbol("");
   491→      toast.success(`${s} added to the watchlist.`);

   514→        method: "PATCH",
   515→        headers: { "Content-Type": "application/json" },
   516→        body: JSON.stringify({ symbol: item.symbol, scan: next }),
   517→      });
   518→    } catch (err) {
   519→      setWatchItems((items) => items.map((x) => (x.symbol === item.symbol ? { ...x, scan: item.scan } : x)));
   520→      toast.error(err instanceof Error ? err.message : "Could not update the scan flag");
   521→    }
   522→  };

   540→    } catch (err) {
   541→      toast.error(err instanceof Error ? err.message : "Clear failed");
   542→    }

   544→
   545→  // ---- derived ----
   546→  const pending = status === "fetching" || status === "thinking";
   547→  const cardSymbol = analyzedSymbol || symbol;
   548→  const openTradeForSymbol = trades?.open.find((t) => t.symbol === symbol) ?? null;
