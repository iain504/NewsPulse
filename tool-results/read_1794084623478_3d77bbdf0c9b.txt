    52→
    53→const SYMBOL_RE = /^[A-Z0-9]{5,15}$/;
    54→
    55→interface SnapshotEvent {
    56→  snapshot: MarketSnapshot;
    57→}
    58→interface SignalEvent {
    59→  signal: StructuredSignal;

    66→
    67→export function SignalsPanel() {
    68→  // ---- market state ----
    69→  const [symbol, setSymbol] = React.useState("BTCUSDT");
    70→  const [timeframe, setTimeframe] = React.useState<Timeframe>("15m");
    71→  const [candles, setCandles] = React.useState<CandleRow[]>([]);
    72→  const [ticker, setTicker] = React.useState<TickerRow | null>(null);
    73→  const [quotes, setQuotes] = React.useState<Record<string, TickerRow>>({});
    74→  const [chartLoading, setChartLoading] = React.useState(true);
    75→  const [chartError, setChartError] = React.useState<string | null>(null);
    76→  const [watchlist, setWatchlist] = React.useState<string[]>([...DEFAULT_SYMBOLS]);
    77→  const [customSymbol, setCustomSymbol] = React.useState("");
    78→
    79→  // ---- analysis state ----
    80→  const [model, setModel] = React.useState("edge0-35b");
    81→  const [riskProfile, setRiskProfile] = React.useState("balanced");
    82→  const [status, setStatus] = React.useState<SignalStatus>("idle");
    83→  const [snapshot, setSnapshot] = React.useState<MarketSnapshot | null>(null);
    84→  const [signal, setSignal] = React.useState<StructuredSignal | null>(null);
    85→  const [rationale, setRationale] = React.useState("");
    86→  const [streaming, setStreaming] = React.useState(false);
    87→  const [analysisError, setAnalysisError] = React.useState<string | null>(null);
    88→  const [latency, setLatency] = React.useState(0);
    89→  const [analyzedKey, setAnalyzedKey] = React.useState("");
    90→
    91→  // ---- auto scan ----
    92→  const [autoOn, setAutoOn] = React.useState(false);
    93→  const [scanMinutes, setScanMinutes] = React.useState<number>(15);
    94→  const [scanSymbols, setScanSymbols] = React.useState<string[]>(["BTCUSDT", "ETHUSDT", "SOLUSDT"]);
    95→  const [nextScanAt, setNextScanAt] = React.useState<number | null>(null);
    96→  const [nowTick, setNowTick] = React.useState(Date.now());
    97→  const [scanProgress, setScanProgress] = React.useState<string | null>(null);
    98→  const runningRef = React.useRef(false);
    99→
   100→  // ---- feed ----
   101→  const [feed, setFeed] = React.useState<SignalRow[]>([]);
   102→  const [feedLoading, setFeedLoading] = React.useState(true);
   103→  const [feedFilter, setFeedFilter] = React.useState("all");
   104→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   105→
   106→  const refreshFeed = React.useCallback(async () => {
   107→    try {
   108→      const res = await fetchJSON<{ ok: boolean; signals: SignalRow[] }>("/api/signals?limit=40");
   109→      setFeed(res.signals);
   110→    } catch {
   111→      // feed is non-critical
   112→    } finally {
   113→      setFeedLoading(false);
   114→    }
   115→  }, []);
   116→
   117→  React.useEffect(() => {
   118→    void refreshFeed();
   119→  }, [refreshFeed]);
   120→
   121→  // restore auto-scan preferences
   122→  React.useEffect(() => {
   123→    try {
   124→      const saved = localStorage.getItem("np.scan");
   125→      if (!saved) return;
   126→      const cfg = JSON.parse(saved) as { minutes?: number; symbols?: string[] };
   127→      if (cfg.minutes && (SCAN_INTERVALS as readonly number[]).includes(cfg.minutes)) {
   128→        setScanMinutes(cfg.minutes);
   129→      }
   130→      if (Array.isArray(cfg.symbols)) {
   131→        setScanSymbols(cfg.symbols.filter((s) => SYMBOL_RE.test(s)).slice(0, 12));
   132→      }
   133→    } catch {
   134→      // ignore malformed storage
   135→    }
   136→  }, []);
   137→
   138→  React.useEffect(() => {
   139→    localStorage.setItem("np.scan", JSON.stringify({ minutes: scanMinutes, symbols: scanSymbols }));
   140→  }, [scanMinutes, scanSymbols]);
   141→
   142→  // ---- chart data (30s refresh) ----
   143→  const loadChart = React.useCallback(async () => {
   144→    setChartLoading(true);
   145→    setChartError(null);
   146→    try {
   147→      const res = await fetchJSON<{ ok: boolean; candles: CandleRow[]; ticker: TickerRow | null }>(
   148→        `/api/market/klines?symbol=${encodeURIComponent(symbol)}&interval=${timeframe}&limit=150`,
   149→      );
   150→      setCandles(res.candles);
   151→      setTicker(res.ticker);
   152→    } catch (err) {
   153→      setChartError(err instanceof Error ? err.message : "Failed to load chart data.");
   154→    } finally {
   155→      setChartLoading(false);
   156→    }
   157→  }, [symbol, timeframe]);
   158→
   159→  React.useEffect(() => {
   160→    void loadChart();
   161→    const iv = setInterval(() => {
   162→      if (document.visibilityState === "visible") void loadChart();
   163→    }, 30_000);
   164→    return () => clearInterval(iv);
   165→  }, [loadChart]);
   166→
   167→  // ---- watchlist quotes (12s refresh) ----
   168→  const loadQuotes = React.useCallback(async () => {
   169→    if (watchlist.length === 0) return;
   170→    try {
   171→      const res = await fetchJSON<{ ok: boolean; tickers: TickerRow[] }>(
   172→        `/api/market/quote?symbols=${encodeURIComponent(watchlist.join(","))}`,
   173→      );
   174→      const map: Record<string, TickerRow> = {};
   175→      for (const t of res.tickers) map[t.symbol] = t;
   176→      setQuotes(map);
   177→    } catch {
   178→      // keep last quotes
   179→    }
   180→  }, [watchlist]);
   181→
   182→  React.useEffect(() => {
   183→    void loadQuotes();
   184→    const iv = setInterval(loadQuotes, 12_000);
   185→    return () => clearInterval(iv);
   186→  }, [loadQuotes]);
   187→
   188→  // ---- SSE analysis ----
   189→  const analyzeSymbol = React.useCallback(
   190→    async (sym: string, source: "manual" | "auto", silent: boolean): Promise<void> => {
   191→      if (!silent) {
   192→        setStatus("fetching");
   193→        setSignal(null);
   194→        setRationale("");
   195→        setStreaming(false);
   196→        setAnalysisError(null);
   197→        setLatency(0);
   198→      }
   199→      let errored = false;
   200→      let errorMessage = "";
   201→      try {
   202→        await postSSE(
   203→          "/api/signals/analyze",
   204→          { symbol: sym, interval: timeframe, model, riskProfile, source },
   205→          {
   206→            onEvent: (event, data) => {
   207→              if (event === "snapshot") {
   208→                if (!silent) {
   209→                  setSnapshot((data as SnapshotEvent).snapshot);
   210→                  setStatus("thinking");
   211→                }
   212→              } else if (event === "signal") {
   213→                if (!silent) {
   214→                  setSignal((data as SignalEvent).signal);
   215→                  setLatency((data as SignalEvent).latencyMs);
   216→                  setStatus("ready");
   217→                  setAnalyzedKey(`${sym}|${timeframe}`);
   218→                }
   219→              } else if (event === "delta") {
   220→                if (!silent) {
   221→                  setStreaming(true);
   222→                  setRationale((r) => r + (data as { t: string }).t);
   223→                }
   224→              } else if (event === "done") {
   225→                if (!silent) setStreaming(false);
   226→                void refreshFeed();
   227→              } else if (event === "error") {
   228→                errored = true;
   229→                errorMessage = (data as ErrorEvent).message;
   230→                if (!silent) {
   231→                  setAnalysisError(errorMessage);
   232→                  setStatus("error");
   233→                  setStreaming(false);
   234→                }
   235→              }
   236→            },
   237→            onError: (err) => {
   238→              errored = true;
   239→              errorMessage = err.message;
   240→              if (!silent) {
   241→                setAnalysisError(err.message);
   242→                setStatus("error");
   243→              }
   244→            },
   245→          },
   246→        );
   247→      } catch (err) {
   248→        errored = true;
   249→        errorMessage = err instanceof Error ? err.message : "Request failed";
   250→        if (!silent) {
   251→          setAnalysisError(errorMessage);
   252→          setStatus("error");
   253→        }
   254→      }
   255→      if (errored) throw new Error(errorMessage || `${sym}: analysis failed`);
   256→    },
   257→    [timeframe, model, riskProfile, refreshFeed],
   258→  );
   259→
   260→  const analyzeNow = React.useCallback(async () => {
   261→    try {
   262→      await analyzeSymbol(symbol, "manual", false);
   263→    } catch {
   264→      // error already reflected in the card
   265→    }
   266→  }, [analyzeSymbol, symbol]);
   267→
   268→  // ---- auto-scan loop ----
   269→  const runScan = React.useCallback(async () => {
   270→    if (runningRef.current) return;
   271→    const list = scanSymbols.length > 0 ? scanSymbols : [symbol];
   272→    runningRef.current = true;
   273→    try {
   274→      for (let i = 0; i < list.length; i++) {
   275→        const sym = list[i];
   276→        setScanProgress(`Analyzing ${sym} (${i + 1}/${list.length})…`);
   277→        try {
   278→          await analyzeSymbol(sym, "auto", true);
   279→        } catch (err) {
   280→          toast.error(`${sym} — ${err instanceof Error ? err.message : "scan failed"}`);
   281→        }
   282→      }
   283→      toast.success(`Auto-scan complete — ${list.length} symbol${list.length === 1 ? "" : "s"} checked.`);
   284→    } finally {
   285→      runningRef.current = false;
   286→      setScanProgress(null);
   287→      setNextScanAt(Date.now() + scanMinutes * 60_000);
   288→    }
   289→  }, [scanSymbols, symbol, scanMinutes, analyzeSymbol]);
   290→
   291→  React.useEffect(() => {
   292→    if (!autoOn) return;
   293→    setNowTick(Date.now());
   294→    const iv = setInterval(() => setNowTick(Date.now()), 1000);
   295→    return () => clearInterval(iv);
   296→  }, [autoOn]);
   297→
   298→  React.useEffect(() => {
   299→    if (!autoOn) return;
   300→    if (nextScanAt === null) {
   301→      setNextScanAt(Date.now() + scanMinutes * 60_000);
   302→      return;
   303→    }
   304→    if (nowTick >= nextScanAt && !runningRef.current) {
   305→      void runScan();
   306→    }
   307→  }, [autoOn, nowTick, nextScanAt, scanMinutes, runScan]);
   308→
   309→  const toggleAuto = (on: boolean) => {
   310→    setAutoOn(on);
   311→    if (on) {
   312→      setNextScanAt(Date.now() + scanMinutes * 60_000);
   313→      toast.info(`Auto-scan armed — every ${scanMinutes} min across ${scanSymbols.length || 1} symbol(s).`);
   314→    } else {
   315→      setNextScanAt(null);
   316→    }
   317→  };
   318→
   319→  // ---- watchlist management ----
   320→  const addSymbol = async () => {
   321→    const s = customSymbol.trim().toUpperCase();
   322→    if (!SYMBOL_RE.test(s)) {
   323→      toast.error("Invalid symbol format — try e.g. LINKUSDT.");
   324→      return;
   325→    }
   326→    if (watchlist.includes(s)) {
   327→      toast.info(`${s} is already on the watchlist.`);
   328→      setCustomSymbol("");
   329→      return;
   330→    }
   331→    try {
   332→      await fetchJSON(`/api/market/klines?symbol=${encodeURIComponent(s)}&interval=15m&limit=60`);
   333→      setWatchlist((w) => [...w, s].slice(0, 12));
   334→      setCustomSymbol("");
   335→      toast.success(`${s} added to the watchlist.`);
   336→    } catch (err) {
   337→      toast.error(`${s} not found on Binance — ${err instanceof Error ? err.message : "check the ticker"}`);
   338→    }
   339→  };
   340→
   341→  const removeSymbol = (s: string) => {
   342→    const next = watchlist.filter((x) => x !== s);
   343→    setWatchlist(next);
   344→    setScanSymbols((prev) => prev.filter((x) => x !== s));
   345→    if (s === symbol && next.length > 0) setSymbol(next[0]);
   346→  };
   347→
   348→  // ---- feed actions ----
   349→  const deleteSignal = async (id: string) => {
   350→    try {
   351→      await fetchJSON(`/api/signals?id=${encodeURIComponent(id)}`, { method: "DELETE" });
   352→      setFeed((f) => f.filter((x) => x.id !== id));
   353→    } catch (err) {
   354→      toast.error(err instanceof Error ? err.message : "Delete failed");
   355→    }
   356→  };
   357→
   358→  const clearAllSignals = async () => {
   359→    if (!window.confirm("Delete all saved signals?")) return;
   360→    try {
   361→      await fetchJSON("/api/signals?all=1", { method: "DELETE" });
   362→      setFeed([]);
   363→      toast.success("Signal history cleared.");
   364→    } catch (err) {
   365→      toast.error(err instanceof Error ? err.message : "Clear failed");
   366→    }
   367→  };
   368→
   369→  // ---- derived ----
   370→  const pending = status === "fetching" || status === "thinking";
   371→  const chartLevels =
   372→    signal && analyzedKey === `${symbol}|${timeframe}` && signal.direction !== "WAIT"
   373→      ? {
   374→          direction: signal.direction,
   375→          entry: signal.entry,
   376→          entryLow: signal.entryLow,
   377→          entryHigh: signal.entryHigh,
   378→          stopLoss: signal.stopLoss,
   379→          takeProfit: signal.takeProfit,
   380→        }
   381→      : null;
   382→
   383→  const remainingMs = nextScanAt !== null ? Math.max(0, nextScanAt - nowTick) : null;
   384→  const countdown =
   385→    remainingMs !== null
   386→      ? `${Math.floor(remainingMs / 60000)}:${Math.floor((remainingMs % 60000) / 1000)
   387→          .toString()
   388→          .padStart(2, "0")}`
   389→      : null;
   390→  const countdownPct =
   391→    remainingMs !== null ? Math.max(0, Math.min(100, (1 - remainingMs / (scanMinutes * 60000)) * 100)) : 0;
   392→
   393→  const filteredFeed = feedFilter === "all" ? feed : feed.filter((s) => s.symbol === feedFilter);
   394→  const counts = feed.reduce(
   395→    (acc, s) => {
   396→      acc[s.direction] = (acc[s.direction] ?? 0) + 1;
   397→      return acc;
   398→    },
   399→    {} as Record<string, number>,
   400→  );
   401→  const snapMatches = snapshot && analyzedKey === `${symbol}|${timeframe}` ? snapshot : null;
   402→
   403→  return (
   404→    <div className="space-y-5">
   405→      {/* intro */}
   406→      <div className="flex flex-wrap items-end justify-between gap-3">
   407→        <div>
   408→          <h2 className="flex items-center gap-2 text-lg font-bold">
   409→            <Activity className="size-5 text-amber-500" aria-hidden />
   410→            Trading Signal Terminal
   411→          </h2>
   412→          <p className="mt-0.5 text-sm text-muted-foreground">
   413→            Live Binance spot bars → deterministic indicators → Edge0 verdict with entry, stop-loss &amp; take-profit.
   414→          </p>
   415→        </div>
   416→        <Badge variant="outline" className="border-amber-300 bg-amber-50 text-[11px] text-amber-800 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-300">
   417→          Educational study tool — not financial advice
   418→        </Badge>
   419→      </div>
   420→
   421→      {/* watchlist strip */}
   422→      <div className="space-y-2">
   423→        <div className="flex items-center justify-between">
   424→          <p className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   425→            <Radio className="size-3.5" aria-hidden /> Watchlist · live 24h
   426→          </p>
   427→          <div className="flex items-center gap-1.5">
   428→            <Input
   429→              value={customSymbol}
   430→              onChange={(e) => setCustomSymbol(e.target.value.toUpperCase())}
   431→              onKeyDown={(e) => {
   432→                if (e.key === "Enter") void addSymbol();
   433→              }}
   434→              placeholder="Add e.g. LINKUSDT"
   435→              className="h-8 w-40 text-xs"
   436→              aria-label="Add symbol to watchlist"
   437→            />
   438→            <Button size="sm" variant="outline" className="h-8 gap-1 px-2" onClick={() => void addSymbol()}>
   439→              <Plus className="size-3.5" aria-hidden /> Add
   440→            </Button>
   441→          </div>
   442→        </div>
   443→        <div className="custom-scrollbar flex gap-2 overflow-x-auto pb-1">
   444→          {watchlist.map((sym) => {
   445→            const q = quotes[sym];
   446→            const active = sym === symbol;
   447→            const up = (q?.priceChangePercent ?? 0) >= 0;
   448→            return (
   449→              <div key={sym} className="relative shrink-0">
   450→                <button
   451→                  type="button"
   452→                  onClick={() => setSymbol(sym)}
   453→                  className={`w-32 rounded-xl border p-2.5 text-left transition-colors ${
   454→                    active
   455→                      ? "border-amber-500 bg-amber-500/10"
   456→                      : "border-border bg-card hover:border-amber-400/60 hover:bg-muted/50"
   457→                  }`}
   458→                  aria-pressed={active}
   459→                  aria-label={`Show ${sym} chart`}
   460→                >
   461→                  <p className="flex items-center gap-1 text-xs font-bold">
   462→                    {sym.replace("USDT", "")}
   463→                    <span className="text-[9px] font-medium text-muted-foreground">/USDT</span>
   464→                    {q && (
   465→                      <span className={`ml-auto text-[10px] font-semibold ${up ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}`}>
   466→                        {formatPct(q.priceChangePercent)}
   467→                      </span>
   468→                    )}
   469→                  </p>
   470→                  <p className="mt-1 font-mono text-sm font-semibold tabular-nums">
   471→                    {q ? formatPrice(q.lastPrice) : <span className="text-muted-foreground">…</span>}
   472→                  </p>
   473→                  <p className="text-[10px] text-muted-foreground">
   474→                    {q ? `Vol ${formatCompact(q.quoteVolume)}` : "loading"}
   475→                  </p>
   476→                </button>
   477→                {watchlist.length > 1 && (
   478→                  <button
   479→                    type="button"
   480→                    onClick={() => removeSymbol(sym)}
   481→                    className="absolute -right-1.5 -top-1.5 hidden rounded-full border bg-background p-0.5 text-muted-foreground hover:text-rose-500 sm:block"
   482→                    aria-label={`Remove ${sym} from watchlist`}
   483→                  >
   484→                    <X className="size-3" aria-hidden />
   485→                  </button>
   486→                )}
   487→              </div>
   488→            );
   489→          })}
   490→        </div>
   491→      </div>
   492→
   493→      {/* chart + engine */}
   494→      <div className="grid gap-4 lg:grid-cols-5">
   495→        {/* chart card */}
   496→        <Card className="lg:col-span-3">
   497→          <CardHeader className="pb-2">
   498→            <div className="flex flex-wrap items-center justify-between gap-2">
   499→              <CardTitle className="flex items-center gap-2 text-base">
   500→                <span className="font-mono">{symbol}</span>
   501→                <span className="text-sm font-normal text-muted-foreground">· Binance spot</span>
   502→                {ticker && (
   503→                  <span className={`text-sm font-bold ${ticker.priceChangePercent >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}`}>
   504→                    {formatPrice(ticker.lastPrice)}{" "}
   505→                    <span className="text-xs font-semibold">{formatPct(ticker.priceChangePercent)}</span>
   506→                  </span>
   507→                )}
   508→              </CardTitle>
   509→              <div className="flex items-center gap-1">
   510→                <div className="flex rounded-lg border p-0.5" role="group" aria-label="Chart timeframe">
   511→                  {TIMEFRAMES.map((tf) => (
   512→                    <button
   513→                      key={tf}
   514→                      type="button"
   515→                      onClick={() => setTimeframe(tf)}
   516→                      className={`rounded-md px-2.5 py-1 text-xs font-medium transition-colors ${
   517→                        timeframe === tf
   518→                          ? "bg-zinc-900 text-amber-300 dark:bg-amber-500 dark:text-zinc-950"
   519→                          : "text-muted-foreground hover:bg-muted"
   520→                      }`}
   521→                      aria-pressed={timeframe === tf}
   522→                    >
   523→                      {tf}
   524→                    </button>
   525→                  ))}
   526→                </div>
   527→                <Button
   528→                  size="icon"
   529→                  variant="ghost"
   530→                  className="size-8"
   531→                  onClick={() => void loadChart()}
   532→                  aria-label="Refresh chart"
   533→                >
   534→                  <RefreshCw className={`size-4 ${chartLoading ? "animate-spin" : ""}`} aria-hidden />
   535→                </Button>
   536→              </div>
   537→            </div>
   538→          </CardHeader>
   539→          <CardContent className="space-y-2">
   540→            {chartError ? (
   541→              <div className="flex h-64 flex-col items-center justify-center gap-2 text-center">
   542→                <p className="text-sm font-medium text-rose-600 dark:text-rose-400">{chartError}</p>
   543→                <Button size="sm" variant="outline" onClick={() => void loadChart()}>
